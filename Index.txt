    <script>

        const popup =
            document.getElementById("videoPopup");

        const popupVideo =
            document.getElementById("popupVideo");

        const contactPopup =
            document.getElementById("contactPopup");

        let previousContactFocus = null;

        /* =========================
           OPEN VIDEO
           ========================= */

        function openVideo(videoFile) {

            popupVideo.pause();

            popupVideo.src = videoFile;

            popup.classList.add("active");

            document.body.style.overflow = "hidden";

            popupVideo.load();

            const playPromise = popupVideo.play();

            if (playPromise !== undefined) {

                playPromise.catch(function(error) {

                    console.log(
                        "Video could not start automatically:",
                        error
                    );

                });
            }
        }

        /* =========================
           CLOSE VIDEO
           ========================= */

        function closeVideo() {

            popupVideo.pause();

            popupVideo.removeAttribute("src");

            popupVideo.load();

            popup.classList.remove("active");

            document.body.style.overflow = "";
        }

        /* =========================
           CLOSE VIDEO OUTSIDE
           ========================= */

        function closeVideoOutside(event) {

            if (event.target === popup) {
                closeVideo();
            }
        }

        /* =========================
           CLOSE WHEN VIDEO ENDS
           ========================= */

        popupVideo.addEventListener("ended", function() {

            console.log("Video finished playing");

            closeVideo();
        });

        /* =========================
           OPEN CONTACT POPUP
           ========================= */

        function openContact() {

            previousContactFocus = document.activeElement;

            contactPopup.classList.add("active");

            document.body.style.overflow = "hidden";

            contactPopup
                .querySelector(".close-button")
                .focus();
        }

        /* =========================
           CLOSE CONTACT POPUP
           ========================= */

        function closeContact() {

            contactPopup.classList.remove("active");

            document.body.style.overflow = "";

            if (previousContactFocus) {
                previousContactFocus.focus();
            }
        }

        /* =========================
           CLOSE CONTACT OUTSIDE
           ========================= */

        function closeContactOutside(event) {

            if (event.target === contactPopup) {
                closeContact();
            }
        }

        /* =========================
           CLOSE WITH ESCAPE
           ========================= */

        document.addEventListener("keydown", function(event) {

            if (event.key === "Escape") {

                if (contactPopup.classList.contains("active")) {
                    closeContact();
                }

                if (popup.classList.contains("active")) {
                    closeVideo();
                }
            }
        });

    </script>
